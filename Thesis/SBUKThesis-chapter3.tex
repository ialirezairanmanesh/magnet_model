\chapter{مطالعه موردی: شرکت‌های پیشرو در امنیت اندروید}\label{chap3}

\section{هدف فصل}
در این فصل، شرکت‌هایی که در حوزه امنیت اپلیکیشن‌ها و دستگاه‌های اندروید فعال هستند، بررسی شده و روش‌ها و ابزارهای آن‌ها تحلیل می‌شوند.

\section{معیارهای انتخاب}
انتخاب شرکت‌ها بر مبنای: پوشش کامل چرخه توسعه \lr{(SAST، DAST، MTD)}، اعتبار جهانی، نوآوری، و امکان استخراج تجربه‌های عملی صورت گرفته است.

\section{مطالعه موردی شرکت‌ها}

\subsection{\lr{NowSecure}}
\lr{NowSecure} یک شرکت آمریکایی فعال در امنیت موبایل است \cite{nowsecure2024}.
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item استفاده از \lr{SAST، DAST، behavioral testing} \cite{nowsecure2024platform}.
  \item ارائه پلتفرمی برای خودکارسازی تست‌ها \lr{(NowSecure Platform)} \cite{nowsecure2024automation}.
  \item همکاری با \lr{Synopsys} برای یکپارچه‌سازی تست نسل بعدی \cite{nowsecure2024synopsys}.
\end{itemize}

\subsection{\lr{Qualys (VMDR Mobile)}}
\lr{Qualys} با افزودن ماژول \lr{VMDR Mobile}، راهکار یکپارچه مدیریت آسیب‌پذیری برای دستگاه‌ها و اپلیکیشن‌ها ارائه می‌دهد \cite{qualys2024vmdr}.
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item انبارش خودکار دستگاه و اپ‌ها، شناسایی \lr{CVE}ها و \lr{misconfiguration}ها \cite{qualys2024inventory}.
  \item تصحیح از راه دور (\lr{OTA})، پچ خودکار، مدیریت وضعیت امنیتی \cite{qualys2024patching}.
  \item ادغام با \lr{MDM/EMM} مانند \lr{Intune} برای ورود خودکار داده \cite{qualys2024mdm}.
\end{itemize}

\subsection{\lr{Checkmarx / Synopsys}}
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item \lr{SAST} قوی در محیط \lr{CI/CD}، تشخیص آسیب‌پذیری‌های کد منبع با پوشش بیش از ۳۵ زبان برنامه‌نویسی \cite{checkmarx2024sast}.
  \item ادغام با ابزارهای تست موبایل از جمله همکاری با \lr{NowSecure} برای پوشش \lr{MAST} \cite{checkmarx2024mobile}.
\end{itemize}

\section{روش‌ها و ابزارهای مشترک}
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item \textbf{تست ایستا (\lr{SAST})}: ابزارهایی مانند \lr{NowSecure} و \lr{Checkmarx} برای اسکن کد در مراحل اولیه توسعه \cite{nowsecure2024sast}.
  \item \textbf{تست پویا (\lr{DAST})}: بررسی رفتار اپ در اجرا، \lr{NowSecure} نمونه‌ای موفق \cite{nowsecure2024dast}.
  \item \textbf{مدیریت آسیب‌پذیری و پچ (\lr{VMDR})}: \lr{Qualys} برای شناسایی و تصحیح خودکار آسیب‌پذیری‌ها \cite{qualys2024vuln}.
  \item \textbf{تحلیل رفتاری و \lr{MTD}}: تشخیص رفتار مخرب در زمان اجرا \lr{(NowSecure و Qualys)} \cite{nowsecure2024mtd}.
  \item \textbf{ادغام \lr{CI/CD} و \lr{DevSecOps}}: خودکارسازی در \lr{pipeline} با \lr{NowSecure}، \lr{Checkmarx} و \lr{Qualys} \cite{checkmarx2024cicd}.
\end{itemize}

\section{مقایسه و تحلیل}
با مقایسه ویژگی‌ها:
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item پوشش چرخه توسعه: به ترتیب \lr{NowSecure}\LTRfootnote{SAST+DAST+behavior}، \lr{Qualys}\LTRfootnote{inventory+vuln+response}، \lr{Checkmarx}\LTRfootnote{SAST}.
  \item امکان اجرا در \lr{pipeline} و سرعت ادغام.
  \item پوشش موبایل گرچه \lr{NowSecure} و \lr{Qualys} کامل‌تر است؛ \lr{Checkmarx} بیشتر بر کد متمرکز است.
\end{itemize}

\section{جمع‌بندی}
هر سه راهکار مزایا و محدودیت دارند:
\begin{itemize}
  \renewcommand{\labelitemi}{\lr{$\bullet$}}
  \item \lr{NowSecure}: کامل‌ترین پوشش، اما نیازمند پیاده‌سازی مستقل.
  \item \lr{Qualys}: مدیریت یکپارچه و مناسب برای سازمان‌های دارای \lr{MDM/EMM}.
  \item \lr{Checkmarx}: قوی در \lr{SAST}، ولی \lr{MAST} را تنها از طریق همکاری‌های خارجی پوشش می‌دهد.
\end{itemize}
این تحلیل کمک می‌کند تا با توجه به مقیاس و امکانات پژوهش یا پروژه، مناسب‌ترین انتخاب انجام شود.
